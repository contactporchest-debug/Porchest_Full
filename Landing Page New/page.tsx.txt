const fadeUp = { hidden: { opacity: 0, y: 22 }, visible: { opacity: 1, y: 0, transition: { duration: 0.55, ease: [0.2, 0.8, 0.2, 1] } } };
const stagger = { visible: { transition: { staggerChildren: 0.08 } } };

function SectionReveal({ children, className }: { children: ReactNode; className?: string }) {
    const ref = useRef<HTMLDivElement>(null);
    const inView = useInView(ref, { once: true, amount: 0.15 });
    return (
        <div ref={ref} className={className}>
            <motion.div initial="hidden" animate={inView ? 'visible' : 'hidden'} variants={stagger}>
                {children}
            </motion.div>
        </div>
    );
}

/* Animated number counter */
function Counter({ from = 0, to, suffix = '', prefix = '', decimals = 0 }: { from?: number; to: number; suffix?: string; prefix?: string; decimals?: number }) {
    const ref = useRef<HTMLSpanElement>(null);
    const inView = useInView(ref, { once: true, amount: 0.5 });
    const mv = useMotionValue(from);
    const rounded = useTransform(mv, latest => `${prefix}${latest.toFixed(decimals)}${suffix}`);
    useEffect(() => {
        if (!inView) return;
        const controls = animate(mv, to, { duration: 1.6, ease: [0.2, 0.8, 0.2, 1] });
        return () => controls.stop();
    }, [inView, mv, to]);
    return <motion.span ref={ref}>{rounded}</motion.span>;
}

function MiniChart() {

                        <stop offset="0%" stopColor="#C2340A" />
                        <stop offset="100%" stopColor="#FF6B1A" />
                    </linearGradient>
                    <linearGradient id="fillGrad" x1="0" y1="0" x2="0" y2="1">

                        <stop offset="100%" stopColor="#C2340A" stopOpacity="0" />
                    </linearGradient>
                </defs>
                {/* grid */}
                {[28, 56, 84, 112].map(y => (
                    <line key={y} x1="0" x2="400" y1={y} y2={y} stroke="#EDD9BC" strokeWidth="1" strokeDasharray="3 4" opacity="0.6" />
                ))}
                {/* area + line */}
                {(() => {
                    const pts = bars.map((b, i) => `${(i / (bars.length - 1)) * 400},${140 - (b / 100) * 130}`).join(' ');

                        <>
                            <polygon points={area} fill="url(#fillGrad)" />
                            <polyline points={pts} fill="none" stroke="url(#lineGrad)" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
                            {bars.map((b, i) => (
                                <circle key={i} cx={(i / (bars.length - 1)) * 400} cy={140 - (b / 100) * 130} r="3" fill="#fff" stroke="#C2340A" strokeWidth="2" />
                            ))}
                        </>
                    );
                })()}
            </svg>
        </div>
    );
}

function BentoCard({ children, span, gridColumn }: { children: ReactNode; span?: string; gridColumn?: string }) {
    const ref = useRef<HTMLDivElement>(null);
    const onMove = (e: React.MouseEvent) => {
        const el = ref.current; if (!el) return;
        const r = el.getBoundingClientRect();
        el.style.setProperty('--mx', `${e.clientX - r.left}px`);
        el.style.setProperty('--my', `${e.clientY - r.top}px`);
    };
    return (
        <div ref={ref} onMouseMove={onMove} className="glass-soft spotlight" style={{
            borderRadius: 18, overflow: 'hidden',
            transition: 'transform .35s cubic-bezier(.2,.8,.2,1), box-shadow .3s',
            gridColumn,
        }}>
            {children}
        </div>
    );
}
